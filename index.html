Algorithm 1: Kahn's Multi-Tier Concurrency Wave Partitioning

Input: Graph G = (V, E)
Output: Concurrency Waves W = [W_0, W_1, ..., W_{k-1}]

1: in_degree ← calculate_in_degrees(G)
2: current_wave ← { v ∈ V | in_degree[v] == 0 }
3: waves ← []
4: while current_wave is not empty do:
5:     waves.append(current_wave)
6:     next_wave ← []
7:     for u in current_wave do:
8:         for v in G.adjacent[u] do:
9:             in_degree[v] ← in_degree[v] - 1
10:            if in_degree[v] == 0 then:
11:                next_wave.append(v)
12:    current_wave ← next_wave
13: return waves

Algorithm 2: Speculative Parallel Dispatch with Deterministic Rollback

Input: P-DAG G, StateManager M, Threshold θ
Output: Canonical State S_canonical

1: waves ← compute_concurrency_waves(G)
2: speculative_pool ← Map()
3: for wave_idx = 0 to len(waves) - 1 do:
4:     // Pre-dispatch next wave across barrier
5:     if wave_idx + 1 < len(waves) then:
6:         for v in waves[wave_idx + 1] do:
7:             if confidence(v) ≥ θ then:
8:                 branch_id ← create_isolated_branch(v)
9:                 X_hat ← predict_inputs(v, M.canonical)
10:                future ← pool.submit(execute_tool, v, X_hat)
11:                speculative_pool[v] ← (future, branch_id, X_hat)
12:    // Execute wave & verify speculative branches
13:    for u in waves[wave_idx] do:
14:        X_actual ← extract_inputs(u, M.canonical)
15:        if u in speculative_pool then:
16:            result ← await future
17:            if X_hat == X_actual then:
18:                M.commit_branch(branch_id, u, result) // 0ms Hit
19:            else:
20:                M.discard_branch(branch_id)           // O(1) Rollback
21:                res ← execute_tool(u, X_actual)
22:                M.commit(u, res)
23: return M.canonical
